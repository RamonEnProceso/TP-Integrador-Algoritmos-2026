- largoEntero(n v, resultado ^)
    - Si n = 0
        - Devolver 1
    - largo <- 0
    - valor <- n
    - Si valor < 0
        - largo <- 1
        - valor <- -valor
    - Mientras valor > 0
        - largo <- largo + 1
        - valor <- valor / 10
    - Devolver largo

- buscarJuegoPorCodigoBinaria(juegos v, n v, codigoBuscado v, resultado ^)
    - inicio <- 0
    - fin <- n - 1
    - Mientras inicio <= fin
        - medio <- inicio + (fin - inicio) / 2
        - Si juegos[medio].codigo = codigoBuscado
            - Devolver medio
        - Sino
            - Si juegos[medio].codigo < codigoBuscado
                - inicio <- medio + 1
            - Sino
                - fin <- medio - 1
    - Devolver -1


- devolverMayorPosV(v v, n v, resultado ^)
    - pos <- 0
    - Para i <- 1 mientras i < n
        - Si v[i] > v[pos]
            - pos <- i
    - Devolver pos

- ordenarDatosMinMax(nombres ^, datos ^, tamaño v)
    - huboCambios <- verdadero
    - Para i <- 0 mientras i < tamaño - 1 y huboCambios
        - huboCambios <- falso
        - Para j -> 0 mientras j < tamaño - i - 1
            - Si datos[j] > datos[j + 1]
                - swapInt(datos[j], datos[j + 1])
                - swapString(nombres[j], nombres[j + 1])
                - huboCambios <- verdadero


- ordenarPorCodigo(juegos ^, n v)
    - huboCambios <- verdadero
    - Para i <- 0 mientras i < n - 1 y huboCambios
        - huboCambios <- falso
        - Para j -> 0 mientras j < n - 1 - i
            - Si juegos[j].codigo > juegos[j + 1].codigo
                - aux <- juegos[j]
                - juegos[j] <- juegos[j + 1]
                - juegos[j + 1] <- aux
                - huboCambios <- verdadero


- compararTextos(a v, b v, resultado ^)
    - i <- 0
    - Mientras a[i] != nulo y b[i] != nulo y a[i] == b[i]
        - i <- i + 1
    - Devolver a[i] - b[i]


- ordenarPorNombre(juegos ^, n v)
    - huboCambios <- verdadero
    - Para i <- 0 mientras i < n - 1 y huboCambios
        - huboCambios <- falso
        - Para j -> 0 mientras j < n - 1 - i
            - Si compararTextos(juegos[j].nombre, juegos[j + 1].nombre) > 0
                - aux <- juegos[j]
                - juegos[j] <- juegos[j + 1]
                - juegos[j + 1] <- aux
                - huboCambios <- verdadero


- aMinuscula(c v, resultado ^)
    - Si c >= 'A' y c <= 'Z'
        - Devolver c + ('a' - 'A')
    - Devolver c


- contieneSubtexto(texto v, subtexto v, resultado ^)
    - i <- 0
    - Mientras texto[i] != nulo
        - j <- 0
        - Mientras subtexto[j] != nulo y texto[i + j] != nulo y texto[i + j] == subtexto[j]
            - j <- j + 1
        - Si subtexto[j] = nulo
            - Devolver verdadero
        - i <- i + 1
    - Devolver falso


- buscarJuegoPorNombre(juegos v, n v, nombreBuscado v, resultado ^)
    - i <- 0
    - Mientras nombreBuscado[i] != nulo y i < LARGO_NOMBRE - 1
        - buscadoMin[i] <- aMinuscula(nombreBuscado[i])
        - i <- i + 1
    - buscadoMin[i] <- nulo
    - Para j -> 0 mientras j < n
        - k <- 0
        - Mientras juegos[j].nombre[k] != nulo y k < LARGO_NOMBRE - 1
            - nombreJuegoMin[k] <- aMinuscula(juegos[j].nombre[k])
            - k <- k + 1
        - nombreJuegoMin[k] <- nulo
        - Si contieneSubtexto(nombreJuegoMin, buscadoMin)
            - Devolver j
    - Devolver -1


- verificarAcceso(juego v, visitante v, resultado ^)
    - Si no juego.operativo
        - Imprimir "[ACCESO DENEGADO] El juego " + juego.nombre + " se encuentra FUERA DE SERVICIO."
        - Devolver falso
    - Si visitante.altura < juego.alturaMinima
        - Imprimir "[ACCESO DENEGADO] El visitante " + visitante.nombre + " no cumple la altura minima requerida."
        - Devolver falso
    - Imprimir "[ACCESO PERMITIDO] " + visitante.nombre + " cumple los requisitos."
    - Devolver verdadero


- atenderSiguienteVisitante(c ^)
    - Si desencolarVisitante(c, v)
        - Imprimir "Atendiendo al visitante: " + v.nombre + " (Pulsera: " + v.pulsera + ")"
    - Sino
        - Imprimir "No hay visitantes en la fila para atender."


- desencolarVisitante(c ^, v ^, resultado ^)
    - Si c.frente = nulo
        - Devolver falso
    - aux <- c.frente
    - v <- aux.visitante
    - c.frente <- aux.siguiente
    - Si c.frente = nulo
        - c.fin <- nulo
    - liberar aux
    - Devolver verdadero


- encolarVisitante(c ^, v v)
    - nuevo <- nuevo nodo visitante
    - nuevo.visitante <- v
    - nuevo.siguiente <- nulo
    - Si c.frente = nulo
        - c.frente <- nuevo
    - Sino
        - c.fin.siguiente <- nuevo
    - c.fin <- nuevo


- estaEnCola(c v, pulsera v, resultado ^)
    - actual <- c.frente
    - Mientras actual != nulo
        - Si actual.visitante.pulsera = pulsera
            - Devolver verdadero
        - actual <- actual.siguiente
    - Devolver falso


- liberarCola(c ^)
    - actual <- c.frente
    - Mientras actual != nulo
        - temp <- actual
        - actual <- actual.siguiente
        - liberar temp
    - c.frente <- nulo
    - c.fin <- nulo

        - aux <- aux.siguiente
        - posicion <- posicion + 1

- buscarBinariaIndice(indice v, cantidad v, pulsera v, resultado ^)
    - inicio <- 0
    - fin <- cantidad - 1
    - Mientras inicio <= fin

    - Devolver -1
